WITH market_stats AS (
    -- Theoretical max minutes in 1.1 years (Dec 2024 - Jan 2026) 
    -- 273 trading days * 390 minutes = 106,470
    SELECT 106470.0 AS expected_minutes
),
gap_calc AS (
    SELECT 
        company_cik,
        t,
        (t - LAG(t) OVER (PARTITION BY company_cik ORDER BY t)) / 60000.0 AS gap_mins
    FROM quotes
),
final_stats AS (
    SELECT 
        company_cik,
        COUNT(*) AS actual_rows,
        MIN(to_timestamp(t/1000)) as first_quote,
        MAX(to_timestamp(t/1000)) as last_quote,
        MAX(gap_mins) as max_gap_mins
    FROM gap_calc
    GROUP BY company_cik
)
SELECT 
    company_cik,
    actual_rows,
    -- Percent of the year actually captured
    ROUND((actual_rows / (SELECT expected_minutes FROM market_stats)) * 100, 2) || '%' as year_coverage,
    -- Percent of the year lost to the single biggest hole
    ROUND((max_gap_mins / (SELECT expected_minutes FROM market_stats)) * 100, 2) || '%' as biggest_gap_impact,
    -- Biggest gap expressed in years
    ROUND(max_gap_mins / (252 * 390), 3) as gap_size_in_years,
    -- Start and End
    first_quote,
    last_quote
FROM final_stats
ORDER BY actual_rows ASC;

WITH market_params AS (
	-- 252 days * 390 minutes = 98,280 expected regular hours rows per year
	-- Total window (Dec 2024 - Jan 2026) is roughly 1.1 years
	SELECT 106470.0 AS theoretical_max_rows
),
stats AS (
	SELECT 
		company_cik,
		COUNT(*) AS actual_rows,
		MIN(to_timestamp(t/1000)) AT TIME ZONE 'UTC' AS start_date,
		MAX(to_timestamp(t/1000)) AT TIME ZONE 'UTC' AS end_date,
		-- Calculate duplicates: Count of minutes where more than one entry exists
		SUM(CASE WHEN dup_count > 1 THEN dup_count ELSE 0 END) AS duplicated_rows,
		MAX(gap_mins) AS max_continuous_gap_mins
	FROM (
		SELECT 
			company_cik, 
			t,
			COUNT(*) OVER (PARTITION BY company_cik, t) AS dup_count,
			(t - LAG(t) OVER (PARTITION BY company_cik ORDER BY t)) / 60000.0 AS gap_mins
		FROM quotes
	) sub
	GROUP BY company_cik
)
SELECT 
	company_cik AS "CIK",
	actual_rows AS "Total_Rows_Found",
	ROUND((actual_rows / (SELECT theoretical_max_rows FROM market_params)) * 100, 2) || '%' AS "Coverage_vs_Market_Open",
	duplicated_rows AS "Corrupted_Duplicate_Rows",
	ROUND((max_continuous_gap_mins / 1440.0), 2) AS "Biggest_Hole_In_Days",
	ROUND(max_continuous_gap_mins / (252.0 * 390.0), 3) AS "Gap_Size_In_Years",
	start_date AS "Data_Starts",
	end_date AS "Data_Ends",
	CASE 
		WHEN actual_rows > (SELECT theoretical_max_rows FROM market_params) * 1.2 THEN 'CRITICAL: Data Inflation/Duplicates'
		WHEN actual_rows < 50000 THEN 'CRITICAL: Pagination Failure'
		WHEN max_continuous_gap_mins > 10000 THEN 'CRITICAL: Temporal Swiss Cheese'
		ELSE 'Nominal'
	END AS "Audit_Status"
FROM stats
ORDER BY "Audit_Status" DESC, actual_rows ASC
