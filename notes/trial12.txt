


WITH market_stats AS (
    -- Theoretical max minutes in 1.1 years (Dec 2024 - Jan 2026) 
    -- 273 trading days * 390 minutes = 106,470
    SELECT 106470.0 AS expected_minutes
),
gap_calc AS (
    SELECT 
        company_cik,
        t,
        (t - LAG(t) OVER (PARTITION BY company_cik ORDER BY t)) / 60000.0 AS gap_mins
    FROM quotes
),
final_stats AS (
    SELECT 
        company_cik,
        COUNT(*) AS actual_rows,
        MIN(to_timestamp(t/1000)) as first_quote,
        MAX(to_timestamp(t/1000)) as last_quote,
        MAX(gap_mins) as max_gap_mins
    FROM gap_calc
    GROUP BY company_cik
)
SELECT 
    company_cik,
    actual_rows,
    -- Percent of the year actually captured
    ROUND((actual_rows / (SELECT expected_minutes FROM market_stats)) * 100, 2) || '%' as year_coverage,
    -- Percent of the year lost to the single biggest hole
    ROUND((max_gap_mins / (SELECT expected_minutes FROM market_stats)) * 100, 2) || '%' as biggest_gap_impact,
    -- Biggest gap expressed in years
    ROUND(max_gap_mins / (252 * 390), 3) as gap_size_in_years,
    -- Start and End
    first_quote,
    last_quote
FROM final_stats
ORDER BY actual_rows ASC;

WITH market_params AS (
    -- Theoretical max for 1-minute bars: ~390 mins/day * 252 trading days/year
    -- Adjust these numbers based on your specific start/end date range
    SELECT 98280 AS theoretical_rows_per_year
),
gap_analysis AS (
    SELECT 
        index_symbol,
        t,
        LAG(t) OVER (PARTITION BY index_symbol ORDER BY t) as prev_t
    FROM index_quotes
),
stats AS (
    SELECT 
        index_symbol,
        COUNT(*) AS actual_rows,
        COUNT(*) - COUNT(DISTINCT t) AS duplicated_rows,
        MAX(t - prev_t) / 60000 AS max_continuous_gap_mins, -- Assumes t is in milliseconds
        TO_TIMESTAMP(MIN(t) / 1000) AS start_date,
        TO_TIMESTAMP(MAX(t) / 1000) AS end_date
    FROM gap_analysis
    GROUP BY index_symbol
)
SELECT 
    index_symbol AS "Ticker",
    actual_rows AS "Total_Rows",
    ((actual_rows::float / (SELECT theoretical_rows_per_year FROM market_params)) * 100, 2) || '%' AS "Coverage_vs_Yearly_Expectation",
    duplicated_rows AS "Corrupted_Duplicates",
    ROUND((max_continuous_gap_mins / 1440.0), 2) AS "Biggest_Hole_In_Days",
    start_date AS "Data_Starts",
    end_date AS "Data_Ends",
    CASE 
        WHEN duplicated_rows > 0 THEN 'CRITICAL: Duplicate Entries'
        WHEN actual_rows < 50000 THEN 'CRITICAL: Pagination Failure'
        WHEN max_continuous_gap_mins > 390 THEN 'WARNING: Full Day Missing'
        WHEN max_continuous_gap_mins > 60 THEN 'MINOR: Intraday Gap'
        ELSE 'Nominal'
    END AS "Audit_Status"
FROM stats
ORDER BY actual_rows ASC;
